def clean_nan(x):
    """Il JSON non ammette NaN: i valori mancanti diventano null."""
    if isinstance(x, dict):
        return {k: clean_nan(v) for k, v in x.items()}
    if isinstance(x, list):
        return [clean_nan(v) for v in x]
    if isinstance(x, float) and x != x:
        return None
    return x
