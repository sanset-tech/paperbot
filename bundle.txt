"""Caricamento e validazione della configurazione."""
import json
from pathlib import Path


class ConfigError(ValueError):
    pass


def load_config(path: str | Path = "config.json") -> dict:
    path = Path(path)
    with path.open(encoding="utf-8") as f:
        cfg = json.load(f)
    cfg["_base_dir"] = str(path.resolve().parent)
    validate(cfg)
    return cfg


def validate(cfg: dict) -> None:
    r, s, c = cfg["risk"], cfg["strategy"], cfg["costs"]
    checks = [
        (len(cfg["universe"]) > 0, "universe vuoto"),
        (0 < r["risk_per_trade"] <= 0.03, "risk_per_trade deve stare tra 0 e 3%"),
        (0 < r["max_position_pct"] <= 1, "max_position_pct fuori range"),
        (0 < r["max_gross_exposure"] <= 1, "niente leva: max_gross_exposure <= 1"),
        (r["max_positions"] >= 1, "max_positions >= 1"),
        (0 < r["drawdown_reduce_at"] < r["drawdown_kill_at"] < 1, "soglie drawdown incoerenti"),
        (s.get("type", "breakout") in ("breakout", "rotation"), "strategy.type: breakout o rotation"),
        (s.get("type") == "rotation" or s["exit_breakout"] < s["entry_breakout"] < s["trend_sma"], "periodi strategia incoerenti"),
        (s.get("type") == "rotation" or (s["initial_stop_atr"] > 0 and s["trailing_stop_atr"] > 0), "stop ATR devono essere > 0"),
        (s.get("type") != "rotation" or s["top_n"] >= 1, "top_n >= 1"),
        (0 <= c.get("cash_rate_fallback", 0) < 0.2, "cash_rate_fallback fuori range"),
        (c["commission_pct"] >= 0 and c["slippage_pct"] >= 0, "costi negativi"),
        (cfg["portfolio"]["initial_capital"] > 0, "capitale iniziale <= 0"),
    ]
    for ok, msg in checks:
        if not ok:
            raise ConfigError(msg)


def resolve(cfg: dict, rel: str) -> Path:
    p = Path(rel)
    return p if p.is_absolute() else Path(cfg.get("_base_dir", ".")) / p

"""Dati di mercato: yfinance (reale), CSV (offline) o sintetico (solo test).

Ogni DataFrame ha indice DatetimeIndex (senza fuso) e colonne
Open, High, Low, Close, Volume, già rettificate per split e dividendi.
"""
from __future__ import annotations

import logging
from pathlib import Path

import numpy as np
import pandas as pd

log = logging.getLogger(__name__)
COLS = ["Open", "High", "Low", "Close", "Volume"]


def clean(df: pd.DataFrame, ticker: str) -> pd.DataFrame:
    df = df.copy()
    df.index = pd.to_datetime(df.index)
    if df.index.tz is not None:
        df.index = df.index.tz_localize(None)
    df.index = df.index.normalize()
    df = df[~df.index.duplicated(keep="last")].sort_index()
    missing = [c for c in COLS if c not in df.columns]
    if missing:
        raise ValueError(f"{ticker}: colonne mancanti {missing}")
    df = df[COLS].astype(float)
    bad = df[["Open", "High", "Low", "Close"]].le(0).any(axis=1) | df[["Open", "High", "Low", "Close"]].isna().any(axis=1)
    if bad.any():
        log.warning("%s: scartate %d barre non valide", ticker, int(bad.sum()))
        df = df[~bad]
    # Coerenza High/Low (dati gratuiti a volte sono sporchi)
    df["High"] = df[["High", "Open", "Close"]].max(axis=1)
    df["Low"] = df[["Low", "Open", "Close"]].min(axis=1)
    df["Volume"] = df["Volume"].fillna(0)
    return df


def load_yfinance(tickers, start, end=None) -> dict[str, pd.DataFrame]:
    try:
        import yfinance as yf
    except ImportError as e:
        raise RuntimeError("yfinance non installato: pip install yfinance") from e
    out = {}
    for t in tickers:
        raw = yf.Ticker(t).history(start=start, end=end, auto_adjust=True, actions=False)
        if raw is None or raw.empty:
            log.error("%s: nessun dato da yfinance, escluso", t)
            continue
        out[t] = clean(raw, t)
    return out


def load_csv(tickers, csv_dir) -> dict[str, pd.DataFrame]:
    out = {}
    for t in tickers:
        p = Path(csv_dir) / f"{t}.csv"
        if not p.exists():
            log.error("%s: file %s mancante, escluso", t, p)
            continue
        out[t] = clean(pd.read_csv(p, index_col=0, parse_dates=True), t)
    return out


def save_csv(data: dict[str, pd.DataFrame], csv_dir) -> None:
    Path(csv_dir).mkdir(parents=True, exist_ok=True)
    for t, df in data.items():
        df.to_csv(Path(csv_dir) / f"{t}.csv")


def synthetic(tickers, start="2010-01-01", end="2024-12-31", seed=42) -> dict[str, pd.DataFrame]:
    """Dati FINTI con regimi di trend, solo per test e demo. Nessun valore predittivo."""
    rng = np.random.default_rng(seed)
    idx = pd.bdate_range(start, end)
    out = {}
    for t in tickers:
        n = len(idx)
        regime = np.repeat(rng.normal(0.0003, 0.0009, n // 150 + 1), 150)[:n]
        vol = np.repeat(rng.uniform(0.007, 0.02, n // 200 + 1), 200)[:n]
        ret = regime + vol * rng.standard_t(4, n) / np.sqrt(2)
        close = 100 * np.exp(np.cumsum(ret))
        gap = rng.normal(0, vol * 0.4)
        open_ = np.r_[close[0], close[:-1]] * np.exp(gap)
        hi = np.maximum(open_, close) * np.exp(np.abs(rng.normal(0, vol * 0.5)))
        lo = np.minimum(open_, close) * np.exp(-np.abs(rng.normal(0, vol * 0.5)))
        df = pd.DataFrame({"Open": open_, "High": hi, "Low": lo, "Close": close,
                           "Volume": rng.integers(1e5, 1e6, n)}, index=idx)
        out[t] = clean(df, t)
    return out


def load(cfg: dict, start=None, end=None) -> dict[str, pd.DataFrame]:
    from .config import resolve
    d = cfg["data"]
    start = start or d["history_start"]
    src = d["source"]
    if src == "yfinance":
        data = load_yfinance(cfg["universe"], start, end)
    elif src == "csv":
        data = load_csv(cfg["universe"], resolve(cfg, d["csv_dir"]))
    elif src == "synthetic":
        data = synthetic(cfg["universe"])
    else:
        raise ValueError(f"sorgente dati sconosciuta: {src}")
    if not data:
        raise RuntimeError("nessun dato caricato")
    return data


def check_staleness(data: dict[str, pd.DataFrame], today: pd.Timestamp, max_days: int) -> list[str]:
    """Ritorna i ticker con dati troppo vecchi (feed rotto, delisting, ecc.)."""
    return [t for t, df in data.items() if (today - df.index[-1]).days > max_days]


def load_cash_rate(cfg: dict, start=None, end=None) -> pd.Series | None:
    """Tasso dei T-bill USA a 13 settimane (^IRX) come proxy del rendimento della liquidità.
    Se non disponibile si usa costs.cash_rate_fallback."""
    if cfg["data"]["source"] != "yfinance" or not cfg["costs"].get("cash_rate_ticker"):
        return None
    try:
        import yfinance as yf
        raw = yf.Ticker(cfg["costs"]["cash_rate_ticker"]).history(start=start or cfg["data"]["history_start"], end=end)
        if raw is None or raw.empty:
            return None
        s = raw["Close"] / 100.0
        s.index = pd.to_datetime(s.index).tz_localize(None).normalize()
        return s[s.between(0, 0.2)].sort_index()
    except Exception as e:  # il tasso è un dettaglio: non deve fermare il bot
        log.warning("tasso liquidità non disponibile (%s): uso il valore di riserva", e)
        return None

"""Strategia trend-following multi-asset (stile Donchian/Turtle con filtro di regime).

Tutti gli indicatori al giorno t usano SOLO dati fino a t compreso.
I segnali calcolati sulla chiusura di t vengono eseguiti all'apertura di t+1.
"""
from __future__ import annotations

import numpy as np
import pandas as pd


def atr(df: pd.DataFrame, n: int) -> pd.Series:
    prev = df["Close"].shift(1)
    tr = pd.concat([df["High"] - df["Low"], (df["High"] - prev).abs(), (df["Low"] - prev).abs()], axis=1).max(axis=1)
    return tr.ewm(alpha=1 / n, adjust=False, min_periods=n).mean()  # media di Wilder


class TrendStrategy:
    def __init__(self, p: dict):
        self.p = p

    @property
    def warmup(self) -> int:
        return max(self.p["trend_sma"], self.p["entry_breakout"], self.p["momentum_lookback"]) + 5

    def prepare(self, df: pd.DataFrame) -> pd.DataFrame:
        p = self.p
        out = df.copy()
        out["sma_trend"] = out["Close"].rolling(p["trend_sma"]).mean()
        # shift(1): il massimo dei N giorni PRECEDENTI, così il breakout è reale e non usa la barra di oggi
        out["hh_entry"] = out["High"].rolling(p["entry_breakout"]).max().shift(1)
        out["ll_exit"] = out["Low"].rolling(p["exit_breakout"]).min().shift(1)
        out["atr"] = atr(out, p["atr_period"])
        out["momentum"] = out["Close"] / out["Close"].shift(p["momentum_lookback"]) - 1
        return out

    @staticmethod
    def _ok(row, cols) -> bool:
        return all(np.isfinite(row[c]) for c in cols)

    def wants_entry(self, row) -> bool:
        if not self._ok(row, ["sma_trend", "hh_entry", "atr", "momentum"]):
            return False
        return row["Close"] > row["sma_trend"] and row["Close"] > row["hh_entry"] and row["momentum"] > 0 and row["atr"] > 0

    def wants_exit(self, row) -> bool:
        if not self._ok(row, ["ll_exit", "sma_trend"]):
            return False
        return row["Close"] < row["ll_exit"] or row["Close"] < row["sma_trend"]

    def initial_stop(self, row) -> float:
        return float(row["Close"] - self.p["initial_stop_atr"] * row["atr"])

    def trail_stop(self, current_stop: float, row) -> float:
        if not np.isfinite(row["atr"]):
            return current_stop
        return max(current_stop, float(row["Close"] - self.p["trailing_stop_atr"] * row["atr"]))  # lo stop sale e basta

    @staticmethod
    def rank_key(row) -> float:
        return float(row["momentum"])


class RotationStrategy:
    """Rotazione mensile (stile Faber / momentum multi-asset).

    Una volta al mese: tra gli ETF sopra la media a 200 giorni e con momentum positivo
    tiene i migliori "top_n". Ogni slot vale 1/top_n del capitale: gli slot vuoti restano
    in liquidità, che matura interessi. Nei mercati ribassisti il portafoglio si svuota da solo.
    """

    def __init__(self, p: dict):
        self.p = p

    @property
    def warmup(self) -> int:
        return max(self.p["trend_sma"], max(self.p["score_lookbacks"]), self.p["vol_lookback"]) + 5

    def prepare(self, df: pd.DataFrame) -> pd.DataFrame:
        p = self.p
        out = df.copy()
        out["sma_trend"] = out["Close"].rolling(p["trend_sma"]).mean()
        out["score"] = sum(out["Close"] / out["Close"].shift(n) - 1 for n in p["score_lookbacks"]) / len(p["score_lookbacks"])
        out["vol"] = out["Close"].pct_change().rolling(p["vol_lookback"]).std() * np.sqrt(252)
        return out

    def eligible(self, row) -> bool:
        vals = [row["sma_trend"], row["score"], row["vol"]]
        if not all(np.isfinite(v) for v in vals):
            return False
        return row["Close"] > row["sma_trend"] and row["score"] > 0 and row["vol"] > 0

    def target_weights(self, rows: dict) -> dict[str, float]:
        p = self.p
        cands = sorted(((t, r) for t, r in rows.items() if self.eligible(r)),
                       key=lambda x: float(x[1]["score"]), reverse=True)[: p["top_n"]]
        if not cands:
            return {}
        slot = 1.0 / p["top_n"]
        if p.get("weighting") == "inverse_vol":
            inv = {t: 1 / float(r["vol"]) for t, r in cands}
            tot = sum(inv.values())
            budget = slot * len(cands)  # gli slot vuoti restano in liquidità
            return {t: budget * v / tot for t, v in inv.items()}
        return {t: slot for t, _ in cands}

"""Gestione del rischio: dimensionamento, limiti di esposizione, circuit breaker sul drawdown."""
from __future__ import annotations

import math


class RiskManager:
    def __init__(self, p: dict, fractional: bool = False):
        self.p = p
        self.fractional = fractional

    def drawdown(self, equity: float, peak: float) -> float:
        return 0.0 if peak <= 0 else max(0.0, 1 - equity / peak)

    def should_kill(self, equity: float, peak: float) -> bool:
        return self.drawdown(equity, peak) >= self.p["drawdown_kill_at"]

    def risk_fraction(self, equity: float, peak: float) -> float:
        r = self.p["risk_per_trade"]
        if self.drawdown(equity, peak) >= self.p["drawdown_reduce_at"]:
            r *= self.p["drawdown_reduce_factor"]
        return r

    def _round(self, q: float) -> float:
        return math.floor(q * 1e4) / 1e4 if self.fractional else float(math.floor(q))

    def size(self, equity: float, peak: float, price: float, stop: float,
             gross_exposure: float, cash_available: float) -> tuple[float, float]:
        """Ritorna (quantità, rischio in valuta). Quantità 0 = non entrare."""
        dist = price - stop
        if price <= 0 or dist <= 0:
            return 0.0, 0.0
        risk_amt = equity * self.risk_fraction(equity, peak)
        q_risk = risk_amt / dist
        q_cap = equity * self.p["max_position_pct"] / price
        room = max(0.0, equity * self.p["max_gross_exposure"] - gross_exposure)
        q_room = min(room, cash_available) / price
        q = self._round(max(0.0, min(q_risk, q_cap, q_room)))
        return q, risk_amt

"""Broker virtuale (paper) con costi, slippage, gap sugli stop e stato persistente su file."""
from __future__ import annotations

import json
import math
import os
import tempfile
from dataclasses import asdict, dataclass, field
from pathlib import Path


@dataclass
class Position:
    ticker: str
    qty: float
    entry_price: float
    entry_date: str
    stop: float
    entry_cost: float = 0.0  # commissioni pagate in entrata


@dataclass
class Order:
    ticker: str
    side: str            # "BUY" | "SELL"
    qty: float
    created: str
    reason: str
    stop: float = 0.0    # solo BUY: stop iniziale (0 = nessuno stop)
    risk_amt: float = 0.0
    add: bool = False    # BUY che incrementa una posizione esistente (ribilanciamento)


@dataclass
class PaperBroker:
    cash: float
    commission_pct: float
    commission_min: float
    slippage_pct: float
    fractional: bool = False
    positions: dict = field(default_factory=dict)
    pending: list = field(default_factory=list)
    trades: list = field(default_factory=list)       # operazioni chiuse
    fills: list = field(default_factory=list)        # tutti gli eseguiti
    equity_curve: list = field(default_factory=list) # [data, equity]
    last_price: dict = field(default_factory=dict)
    peak_equity: float = 0.0
    halted: bool = False
    last_processed: str | None = None
    last_rebalance: str | None = None
    interest_earned: float = 0.0

    def accrue_interest(self, annual_rate: float, days: float) -> None:
        """Interessi sulla liquidità (come un conto deposito / BOT a breve)."""
        if self.cash > 0 and annual_rate > 0 and days > 0:
            i = self.cash * annual_rate * days / 365.0
            self.cash += i
            self.interest_earned += i

    # ---------- costi ----------
    def commission(self, notional: float) -> float:
        return max(self.commission_min, abs(notional) * self.commission_pct)

    def _round(self, q):
        return math.floor(q * 1e4) / 1e4 if self.fractional else float(math.floor(q))

    # ---------- ordini ----------
    def submit(self, order: Order) -> None:
        if order.qty <= 0:
            return
        if any(o.ticker == order.ticker and o.side == order.side for o in self.pending):
            return
        self.pending.append(order)

    def pending_buy_notional(self, prices: dict) -> float:
        return sum(o.qty * prices.get(o.ticker, 0.0) for o in self.pending if o.side == "BUY")

    def fill_pending(self, date: str, bars: dict) -> None:
        """Esegue gli ordini all'APERTURA della barra di oggi (generati ieri in chiusura)."""
        keep = []
        # prima le vendite: liberano liquidità per gli acquisti
        for o in sorted(self.pending, key=lambda o: o.side != "SELL"):
            bar = bars.get(o.ticker)
            if bar is None:
                keep.append(o)  # mercato chiuso per questo strumento: riprova domani
                continue
            if o.side == "SELL":
                if o.ticker in self.positions:
                    self._close(o.ticker, bar["Open"] * (1 - self.slippage_pct), date, o.reason, o.qty)
            else:
                self._open(o, bar, date)
        self.pending = keep

    def _open(self, o: Order, bar, date: str) -> None:
        if o.ticker in self.positions and not o.add:
            return
        px = bar["Open"] * (1 + self.slippage_pct)
        if px <= o.stop:
            self._log_fill(date, o.ticker, "SKIP", 0, px, 0, "apertura sotto lo stop: ingresso annullato")
            return
        qty = o.qty
        if o.risk_amt > 0:  # se c'è gap in su il rischio per azione cresce: ridimensiona
            qty = min(qty, self._round(o.risk_amt / (px - o.stop)))
        while qty > 0 and qty * px + self.commission(qty * px) > self.cash:
            qty = self._round(qty * 0.98) if self.fractional else qty - 1
        if qty <= 0:
            self._log_fill(date, o.ticker, "SKIP", 0, px, 0, "liquidità insufficiente")
            return
        fee = self.commission(qty * px)
        self.cash -= qty * px + fee
        if o.ticker in self.positions:  # incremento: prezzo medio di carico
            p = self.positions[o.ticker]
            p.entry_price = (p.entry_price * p.qty + px * qty) / (p.qty + qty)
            p.qty += qty
            p.entry_cost += fee
        else:
            self.positions[o.ticker] = Position(o.ticker, qty, px, date, o.stop, fee)
        self._log_fill(date, o.ticker, "BUY", qty, px, fee, o.reason)

    def _close(self, ticker: str, px: float, date: str, reason: str, qty: float | None = None) -> None:
        p: Position = self.positions[ticker]
        q = p.qty if qty is None or qty >= p.qty else qty
        fee = self.commission(q * px)
        share = q / p.qty
        entry_cost = p.entry_cost * share
        self.cash += q * px - fee
        pnl = (px - p.entry_price) * q - fee - entry_cost
        if q >= p.qty:
            del self.positions[ticker]
        else:
            p.qty -= q
            p.entry_cost -= entry_cost
        self.trades.append({
            "ticker": ticker, "entry_date": p.entry_date, "exit_date": date,
            "qty": q, "entry": round(p.entry_price, 4), "exit": round(px, 4),
            "pnl": round(pnl, 2), "ret_pct": round(100 * pnl / (p.entry_price * q), 2), "reason": reason,
        })
        self._log_fill(date, ticker, "SELL", q, px, fee, reason)

    def check_stops(self, date: str, bars: dict, skip: set | None = None) -> None:
        """Stop eseguiti intraday. Se l'apertura è già sotto lo stop (gap), esce all'apertura: perdita reale, non teorica."""
        for t in list(self.positions):
            bar = bars.get(t)
            if bar is None:
                continue
            stop = self.positions[t].stop
            if stop <= 0:
                continue
            if bar["Open"] <= stop:
                self._close(t, bar["Open"] * (1 - self.slippage_pct), date, "stop (gap in apertura)")
            elif bar["Low"] <= stop:
                self._close(t, stop * (1 - self.slippage_pct), date, "stop")
        # un SELL in coda per una posizione già chiusa dallo stop non serve più
        self.pending = [o for o in self.pending if not (o.side == "SELL" and o.ticker not in self.positions)]

    def mark(self, date: str, bars: dict) -> float:
        for t, bar in bars.items():
            self.last_price[t] = float(bar["Close"])
        eq = self.equity()
        self.peak_equity = max(self.peak_equity, eq)
        self.equity_curve.append([date, round(eq, 2)])
        return eq

    def equity(self) -> float:
        return self.cash + sum(p.qty * self.last_price.get(t, p.entry_price) for t, p in self.positions.items())

    def gross_exposure(self) -> float:
        return sum(p.qty * self.last_price.get(t, p.entry_price) for t, p in self.positions.items())

    def _log_fill(self, date, ticker, side, qty, px, fee, reason):
        self.fills.append({"date": date, "ticker": ticker, "side": side, "qty": qty,
                           "price": round(px, 4), "fee": round(fee, 2), "reason": reason})

    # ---------- persistenza ----------
    def to_dict(self) -> dict:
        d = asdict(self)
        d["positions"] = {t: asdict(p) for t, p in self.positions.items()}
        d["pending"] = [asdict(o) for o in self.pending]
        return d

    @classmethod
    def from_dict(cls, d: dict) -> "PaperBroker":
        d = dict(d)
        pos = {t: Position(**p) for t, p in d.pop("positions", {}).items()}
        pend = [Order(**o) for o in d.pop("pending", [])]
        b = cls(**d)
        b.positions, b.pending = pos, pend
        return b

    def save(self, path: str | Path) -> None:
        """Scrittura atomica: se il processo muore a metà, il vecchio stato resta intatto."""
        path = Path(path)
        path.parent.mkdir(parents=True, exist_ok=True)
        fd, tmp = tempfile.mkstemp(dir=path.parent, suffix=".tmp")
        with os.fdopen(fd, "w", encoding="utf-8") as f:
            json.dump(self.to_dict(), f, indent=1)
        os.replace(tmp, path)

    @classmethod
    def load(cls, path: str | Path) -> "PaperBroker":
        with open(path, encoding="utf-8") as f:
            return cls.from_dict(json.load(f))

def load_blackout(path: Path) -> set[str]:
    """Date (YYYY-MM-DD) in cui NON aprire nuove posizioni: FOMC, BCE, NFP, CPI...
    Un file compilato a mano è più affidabile di uno scraping che si rompe."""
    if not path.exists():
        return set()
    df = pd.read_csv(path, comment="#")
    return set(pd.to_datetime(df.iloc[:, 0]).dt.strftime("%Y-%m-%d"))


class Engine:
    def __init__(self, cfg: dict, data: dict[str, pd.DataFrame], broker: PaperBroker,
                 blackout: set[str] | None = None, cash_rate: pd.Series | None = None):
        self.cfg = cfg
        self.kind = cfg["strategy"].get("type", "breakout")
        self.strategy = RotationStrategy(cfg["strategy"]) if self.kind == "rotation" else TrendStrategy(cfg["strategy"])
        self.cash_rate = cash_rate
        self.cash_rate_fallback = cfg["costs"].get("cash_rate_fallback", 0.0)
        self.risk = RiskManager(cfg["risk"], cfg["portfolio"]["fractional_shares"])
        self.data = {t: self.strategy.prepare(df) for t, df in data.items()}
        self.broker = broker
        self.blackout = blackout or set()

    def calendar(self) -> pd.DatetimeIndex:
        return pd.DatetimeIndex(sorted(set().union(*[df.index for df in self.data.values()])))

    def bars_on(self, date: pd.Timestamp) -> dict:
        return {t: df.loc[date] for t, df in self.data.items() if date in df.index}

    def step(self, date: pd.Timestamp) -> list[str]:
        """Un giorno di mercato. Ritorna un elenco di eventi leggibili."""
        b, ds = self.broker, date.strftime("%Y-%m-%d")
        bars = self.bars_on(date)
        n_fills = len(b.fills)

        b.fill_pending(ds, bars)       # 1. ordini di ieri all'apertura
        b.check_stops(ds, bars)        # 2. stop intraday
        if b.last_processed:           # interessi sulla liquidità per i giorni trascorsi
            b.accrue_interest(self.rate_on(date), (date - pd.Timestamp(b.last_processed)).days)
        eq = b.mark(ds, bars)          # 3. valorizzazione in chiusura
        b.last_processed = ds

        # 4. circuit breaker: drawdown oltre soglia => liquida tutto e fermati
        if not b.halted and self.risk.should_kill(eq, b.peak_equity):
            b.halted = True
            log.critical("KILL SWITCH: drawdown %.1f%%. Liquidazione e stop operatività.",
                         100 * self.risk.drawdown(eq, b.peak_equity))
        if b.halted:
            b.pending = [o for o in b.pending if o.side == "SELL"]
            for t, p in b.positions.items():
                b.submit(Order(t, "SELL", p.qty, ds, "kill switch"))
            return self._events(n_fills)

        if self.kind == "rotation":
            self._rotation(date, ds, bars, eq)
            return self._events(n_fills)

        # 5. uscite e trailing stop
        for t, p in list(b.positions.items()):
            row = bars.get(t)
            if row is None:
                continue
            if self.strategy.wants_exit(row):
                b.submit(Order(t, "SELL", p.qty, ds, "uscita: rottura minimi/trend"))
            else:
                p.stop = self.strategy.trail_stop(p.stop, row)

        # 6. nuovi ingressi
        if ds in self.blackout:
            log.info("%s: data in blackout macro, nessun nuovo ingresso", ds)
            return self._events(n_fills)

        selling = {o.ticker for o in b.pending if o.side == "SELL"}
        held = set(b.positions) - selling
        pending_buys = {o.ticker for o in b.pending if o.side == "BUY"}
        slots = self.cfg["risk"]["max_positions"] - len(held) - len(pending_buys)
        if slots <= 0:
            return self._events(n_fills)

        cands = [(t, r) for t, r in bars.items()
                 if t not in b.positions and t not in pending_buys and self.strategy.wants_entry(r)]
        cands.sort(key=lambda x: self.strategy.rank_key(x[1]), reverse=True)

        gross = b.gross_exposure() + b.pending_buy_notional(b.last_price)
        cash = b.cash - b.pending_buy_notional(b.last_price)
        for t, row in cands[:slots]:
            px = float(row["Close"])
            stop = self.strategy.initial_stop(row)
            qty, risk_amt = self.risk.size(eq, b.peak_equity, px, stop, gross, cash)
            if qty <= 0:
                continue
            b.submit(Order(t, "BUY", qty, ds, "breakout sopra trend", stop=stop, risk_amt=risk_amt))
            gross += qty * px
            cash -= qty * px * (1 + b.slippage_pct) + b.commission(qty * px)
        return self._events(n_fills)

    def rate_on(self, date: pd.Timestamp) -> float:
        if self.cash_rate is not None and len(self.cash_rate):
            v = self.cash_rate.asof(date)
            if pd.notna(v):
                return max(0.0, float(v))
        return self.cash_rate_fallback

    def _rotation(self, date: pd.Timestamp, ds: str, bars: dict, eq: float) -> None:
        """Ribilancia il primo giorno di borsa di ogni mese (ordini eseguiti all'apertura successiva)."""
        b, r = self.broker, self.cfg["risk"]
        if b.last_rebalance and pd.Timestamp(b.last_rebalance).to_period("M") == date.to_period("M"):
            return
        # serve il dato di oggi per tutto l'universo, altrimenti si rimanda a domani
        if len(bars) < len(self.data) * 0.8:
            log.warning("%s: dati incompleti, ribilanciamento rimandato", ds)
            return
        weights = self.strategy.target_weights(bars)
        if ds in self.blackout:
            weights = {t: w for t, w in weights.items() if t in b.positions}  # niente nuovi ingressi
        if self.risk.drawdown(eq, b.peak_equity) >= r["drawdown_reduce_at"]:
            weights = {t: w * r["drawdown_reduce_factor"] for t, w in weights.items()}
        weights = {t: min(w, r["max_position_pct"]) for t, w in weights.items()}
        tot = sum(weights.values())
        if tot > r["max_gross_exposure"]:
            weights = {t: w * r["max_gross_exposure"] / tot for t, w in weights.items()}

        investable = eq * (1 - 2 * (b.slippage_pct + b.commission_pct))  # margine per i costi
        band = self.cfg["strategy"].get("rebalance_band", 0.02) * eq
        rnd = self.risk._round
        b.pending = [o for o in b.pending if o.side == "SELL"]
        for t in set(b.positions) | set(weights):
            px = b.last_price.get(t)
            if not px:
                continue
            cur = b.positions[t].qty if t in b.positions else 0.0
            tgt = rnd(investable * weights.get(t, 0.0) / px)
            delta = tgt - cur
            if delta == 0:
                continue
            full_change = cur == 0 or tgt == 0
            if not full_change and abs(delta) * px < band:
                continue  # differenza piccola: non vale le commissioni
            if delta < 0:
                why = "rotazione: esce" if tgt == 0 else "rotazione: riduce"
                b.submit(Order(t, "SELL", -delta, ds, why))
            else:
                why = "rotazione: entra" if cur == 0 else "rotazione: aumenta"
                b.submit(Order(t, "BUY", delta, ds, why, add=cur > 0))
        b.last_rebalance = ds
        log.info("%s ribilanciamento: %s", ds, {t: round(w, 3) for t, w in sorted(weights.items())} or "tutto liquidità")

    def _events(self, since: int) -> list[str]:
        return [f"{f['date']} {f['side']} {f['ticker']} x{f['qty']} @ {f['price']} ({f['reason']})"
                for f in self.broker.fills[since:]]

"""Metriche di performance."""
from __future__ import annotations

import numpy as np
import pandas as pd


def equity_series(curve: list) -> pd.Series:
    s = pd.Series({pd.Timestamp(d): v for d, v in curve})
    return s.sort_index()


def summary(eq: pd.Series, trades: list | None = None) -> dict:
    eq = eq.dropna()
    if len(eq) < 2:
        return {}
    rets = eq.pct_change().dropna()
    years = (eq.index[-1] - eq.index[0]).days / 365.25
    cagr = (eq.iloc[-1] / eq.iloc[0]) ** (1 / years) - 1 if years > 0 else 0.0
    vol = rets.std() * np.sqrt(252)
    downside = rets[rets < 0].std() * np.sqrt(252)
    dd = eq / eq.cummax() - 1
    out = {
        "inizio": eq.index[0].date().isoformat(),
        "fine": eq.index[-1].date().isoformat(),
        "capitale_finale": round(float(eq.iloc[-1]), 2),
        "rendimento_totale_%": round(100 * (eq.iloc[-1] / eq.iloc[0] - 1), 2),
        "CAGR_%": round(100 * cagr, 2),
        "volatilita_%": round(100 * vol, 2),
        "sharpe": round(rets.mean() / rets.std() * np.sqrt(252), 2) if rets.std() > 0 else 0.0,
        "sortino": round(rets.mean() * 252 / downside, 2) if downside > 0 else 0.0,
        "max_drawdown_%": round(100 * dd.min(), 2),
        "calmar": round(cagr / abs(dd.min()), 2) if dd.min() < 0 else 0.0,
    }
    if trades:
        pnl = np.array([t["pnl"] for t in trades])
        wins, losses = pnl[pnl > 0], pnl[pnl <= 0]
        out.update({
            "operazioni": len(pnl),
            "win_rate_%": round(100 * len(wins) / len(pnl), 1),
            "profit_factor": round(wins.sum() / abs(losses.sum()), 2) if losses.sum() != 0 else float("inf"),
            "media_vincita": round(float(wins.mean()), 2) if len(wins) else 0.0,
            "media_perdita": round(float(losses.mean()), 2) if len(losses) else 0.0,
        })
    return out


def buy_and_hold(data: dict[str, pd.DataFrame], start, end, capital: float) -> pd.Series:
    """Benchmark: stesso universo, pesi uguali, comprato e tenuto. Se il bot non batte questo, non serve."""
    closes = pd.DataFrame({t: df["Close"] for t, df in data.items()}).loc[start:end].ffill().dropna(axis=1, how="all")
    closes = closes.bfill()
    norm = closes / closes.iloc[0]
    return norm.mean(axis=1) * capital

def last_complete_day(now_rome: datetime, after_hour: int) -> pd.Timestamp:
    """La barra di oggi è considerata chiusa solo dopo l'ora indicata (ora di Roma)."""
    today = pd.Timestamp(now_rome.date())
    return today if now_rome.hour >= after_hour else today - pd.Timedelta(days=1)


def status(b: PaperBroker) -> str:
    eq = b.equity()
    lines = [f"Ultimo giorno processato: {b.last_processed}",
             f"Equity: {eq:,.2f} | Cassa: {b.cash:,.2f} | Picco: {b.peak_equity:,.2f} | "
             f"Drawdown: {100 * (1 - eq / b.peak_equity) if b.peak_equity else 0:.1f}%",
             f"Interessi maturati sulla liquidità: {b.interest_earned:,.2f}",
             f"Ultimo ribilanciamento: {b.last_rebalance or '-'}",
             f"Bloccato (kill switch): {'SI' if b.halted else 'no'}", "Posizioni:"]
    for t, p in b.positions.items():
        px = b.last_price.get(t, p.entry_price)
        stop = f"stop {p.stop:>9.2f} " if p.stop > 0 else ""
        lines.append(f"  {t:6} qty {p.qty:>8} carico {p.entry_price:>9.2f} ultimo {px:>9.2f} "
                     f"{stop}peso {100 * p.qty * px / eq:4.1f}% P&L {100 * (px / p.entry_price - 1):+.1f}%")
    if not b.positions:
        lines.append("  (nessuna)")
    lines.append("Ordini per la prossima apertura:")
    for o in b.pending:
        lines.append(f"  {o.side} {o.ticker} x{o.qty} ({o.reason})")
    if not b.pending:
        lines.append("  (nessuno)")
    return "\n".join(lines)


def append_csv(path: Path, rows: list[dict]):
    if not rows:
        return
    new = not path.exists()
    with path.open("a", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, fieldnames=list(rows[0].keys()))
        if new:
            w.writeheader()
        w.writerows(rows)

def run(cfg, data, start=None, end=None, cash_rate=None):

def fmt(x, d=2):
    return f"{x:,.{d}f}".replace(",", "X").replace(".", ",").replace("X", ".")


def spark(curve, w=340, h=120):
    vals = [v for _, v in curve][-260:]
    if len(vals) < 2:
        return '<p class="muted">Il grafico compare dopo qualche giorno di dati.</p>'
    lo, hi = min(vals), max(vals)
    rng = (hi - lo) or 1
    pts = " ".join(f"{i * w / (len(vals) - 1):.1f},{h - 6 - (v - lo) / rng * (h - 12):.1f}" for i, v in enumerate(vals))
    color = "#2e7d32" if vals[-1] >= vals[0] else "#c62828"
    return (f'<svg viewBox="0 0 {w} {h}" class="chart" preserveAspectRatio="none">'
            f'<polyline fill="none" stroke="{color}" stroke-width="2.5" points="{pts}"/></svg>'
            f'<div class="axis"><span>{fmt(lo, 0)}</span><span>{fmt(hi, 0)}</span></div>')


def health(state, now_rome):
    if state.get("halted"):
        return "ALLARME", "Kill switch scattato: il bot ha liquidato e si è fermato.", "bad"
    lp = state.get("last_processed")
    if not lp:
        return "ATTENZIONE", "Nessun giorno ancora processato.", "warn"
    days = (now_rome.date() - datetime.strptime(lp, "%Y-%m-%d").date()).days
    if days > 5:
        return "ALLARME", f"Ultimo giorno processato {days} giorni fa: il bot non sta girando.", "bad"
    eq, peak = equity(state), state.get("peak_equity") or 1
    dd = 1 - eq / peak
    if dd >= 0.12:
        return "ATTENZIONE", f"Drawdown {fmt(100 * dd, 1)}%: vicino alle soglie di protezione.", "warn"
    return "OK", "Il bot gira regolarmente.", "ok"


def equity(s):
    return s["cash"] + sum(p["qty"] * s["last_price"].get(t, p["entry_price"]) for t, p in s["positions"].items())


def build(cfg, out_dir: Path):

    state = json.loads(sp.read_text(encoding="utf-8")) if sp.exists() else None

    if state is None:

        peak = state.get("peak_equity") or eq
        tot = 100 * (eq / cap - 1)
        dd = 100 * (1 - eq / peak) if peak else 0
        lvl, msg, cls = health(state, now)
        rows = ""
        for t, p in sorted(state["positions"].items(), key=lambda x: -x[1]["qty"] * state["last_price"].get(x[0], 0)):

            val = p["qty"] * px
            pnl = 100 * (px / p["entry_price"] - 1)
            rows += (f'<tr><td><b>{t}</b><br><span class="muted">{NAMES.get(t, "")}</span></td>'
